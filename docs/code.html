## What's New in v3.0.0 (Enterprise Air-Gapped Clinical Production)
1. **Zero-Internet Offline SQLite WAL Store (`src/offline_reference_store.py`):** 9,249 indexed GRCh38 records for `--network=none` hospital execution (`gec-audit --offline`).
2. **Sample Zygosity (`GT`/`DP`/`GQ`) & AIM Ancestry Inference (`src/vcf_patient_profiler.py`):** Distinguishes `AR_Carrier_Only` vs `AR_Biallelic_Affected` (e.g., `HBB_E7V` `0/1` vs `1/1`) and infers 7-ancestry patient background (`--default-ancestry auto`).
3. **Constant-Memory Streaming WES Auditor & Zero-Dependency OpenAPI Server (`src/audit_vcf.py`, `src/api_server.py`):** Streams `.vcf`/`.vcf.gz` (10,000-variant WES benchmarked `<5s`, `<250 MB` RAM) with `GET /health`, `/openapi.json`, `/docs`, `POST /audit`, and `POST /audit-vcf`.
4. **NABL (ISO 15189) / CAP / EU AI Act Annex IV Clinical QA HTML Report (`gec-report`):** Offline print-ready patient report with SHA-256 cryptographic traceability hashes.

## What's New in v2.0.0
1. **Autonomous Unlisted Founder Discovery (`src/discover_unlisted_ba1_candidates.py`):** Discovered **19 unlisted BS1+ pathogenic founder/balancing-selection variants** (`ABCA4_G1961E` ASJ 2.39%, `ATP7B_H1069Q` ASJ, `SLC12A3_T180K` EAS, `SLC26A4_F335L` & `SLC26A4_V239D` SAS, etc.) saved in `artifacts/v20_novel_founder_discoveries.json` & `.csv`.
2. **Global CPIC Pharmacogenomics Sidecar (`src/pgx_drug_safety.py`):** Drug-toxicity alerts for `DPYD`, `NUDT15`, `TPMT`, `CYP2C19`, `SLCO1B1`, `G6PD`, and `HLA-B*57:01`/`B*15:02`, written directly into annotated VCF 4.2 (`GEC_PGX_ALERT`).
3. **AlphaFold DB v6 C-Beta RSA Burial Upgrade (`src/v2_mavedb_rsa_upgrade.py`):** Flipped `TP53` MaveDB DMS from miss to **WIN** (`0.2485 -> 0.2488`), alongside wins on `BRCA1`, `PTEN`, and `MSH2` (`LDLR` `-0.0002` honestly reported).

def empirical_lr_bootstrap(y, scores, threshold, direction="ge", n_boot=2000, seed=42):
    """Computes empirical LR+ and 1-sided 95% conservative bootstrap bound (Pejaver 2022)."""
    y = np.asarray(y, dtype=int)
    scores = np.asarray(scores, dtype=float)
    rng = np.random.default_rng(seed)
    n = len(y)
    lrs = []
    for _ in range(n_boot):
        idx = rng.integers(0, n, size=n)
        yb, sb = y[idx], scores[idx]
        pos, neg = (yb == 1), (yb == 0)
        if np.sum(pos) == 0 or np.sum(neg) == 0: continue
        hit = (sb >= threshold) if direction == "ge" else (sb <= threshold)
        # Laplace smoothing (+0.5) prevents divide-by-zero on extreme tails
        tpr = (np.sum(hit & pos) + 0.5) / (np.sum(pos) + 1.0)
        fpr = (np.sum(hit & neg) + 0.5) / (np.sum(neg) + 1.0)

    else:
        # Benign BP4: conservative = 95th percentile upper bound (must stay <= target)

class LocalizedWeightedConformal:
    """
    Localized Weighted Split Conformal Classifier (Tibshirani 2019 + Guan 2023).
    Computes continuous covariate-weighted conformal quantiles over log-AF vectors
    with a guaranteed non-empty prediction set floor.
    """
    def __init__(self, alpha=0.10, bandwidth=1.5):
        self.alpha = alpha
        self.bandwidth = bandwidth
        self.z_cal = None
        self.scores_cal = None

    def fit(self, probs_pos, y_true, z_cov):
        probs_pos = np.asarray(probs_pos, dtype=float)
        y_true = np.asarray(y_true, dtype=int)
        self.z_cal = np.asarray(z_cov, dtype=float)
        p_true = np.where(y_true == 1, probs_pos, 1.0 - probs_pos)
        self.scores_cal = 1.0 - p_true
        return self

    def _weighted_quantile(self, scores, weights, q_level):
        order = np.argsort(scores)
        s_sorted = scores[order]
        w_sorted = weights[order]
        cdf = np.cumsum(w_sorted) / np.sum(w_sorted)
        idx = np.searchsorted(cdf, q_level, side="left")
        return float(s_sorted[min(idx, len(s_sorted) - 1)])

    def predict_sets(self, probs_pos, z_query):
        probs_pos = np.asarray(probs_pos, dtype=float)
        z_query = np.asarray(z_query, dtype=float)
        # Vectorized Gaussian kernel weights in continuous log-AF space
        diff = z_query[:, None, :] - self.z_cal[None, :, :]
        dist2 = np.sum(diff * diff, axis=2)
        W = np.exp(-dist2 / (2.0 * (self.bandwidth ** 2))) + 1e-8

        pred_sets = []
        target = 1.0 - self.alpha
        for i, p in enumerate(probs_pos):
            w = W[i]
            # Append point mass 1.0 at infinity (finite-sample weighted conformal correction)
            w_norm = w / (np.sum(w) + 1.0)
            q = self._weighted_quantile(self.scores_cal, w_norm, min(0.9999, target))
            s = []
            if p <= q: s.append(0)
            if (1.0 - p) <= q: s.append(1)
            if not s: s.append(1 if p >= 0.5 else 0)
            pred_sets.append(s)
        return pred_sets

import numpy as np

def compute_ece(y_true, y_prob, n_bins=10):
    y_true = np.asarray(y_true, dtype=float)
    y_prob = np.asarray(y_prob, dtype=float)
    bins = np.linspace(0.0, 1.0, n_bins + 1)
    ece, n = 0.0, len(y_true)
    if n == 0:
        return 0.0
    for i in range(n_bins):
        mask = (y_prob >= bins[i]) & (y_prob < bins[i + 1] if i < n_bins - 1 else y_prob <= bins[i + 1])
        if np.any(mask):
            ece += (np.sum(mask) / n) * abs(np.mean(y_true[mask]) - np.mean(y_prob[mask]))
    return float(ece)

class AncestryMondrianConformal:
    """Non-empty Ancestry-Conditional Mondrian Split Conformal Classifier (v1.0.0)."""
    def __init__(self, alpha=0.10):

        self.q_by_group = {}
        self.q_global = 1.0

    def _nonconformity(self, probs_pos, y_true):

    def _finite_quantile(self, scores):

        if n == 0:
            return 1.0
        level = min(1.0, np.ceil((n + 1) * (1.0 - self.alpha)) / n)

def kmer_set(seq, k=5):
    return {seq[i:i+k] for i in range(max(0, len(seq) - k + 1))}

def main():
    """Zero-dependency <30% homology clustering via k-mer Jaccard + Union-Find DSA."""
    gene_seqs = {}
    with open("data/v1_clinvar_120genes.csv") as f:
        for r in csv.DictReader(f):
            if r["gene"] not in gene_seqs:
                gene_seqs[r["gene"]] = r["seq"]

    genes = sorted(gene_seqs.keys())
    kmers = {g: kmer_set(gene_seqs[g], k=5) for g in genes}
    parent = {g: g for g in genes}

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb: parent[ra] = rb

    merged_pairs = []
    for i in range(len(genes)):
        for j in range(i + 1, len(genes)):

            if not k1 or not k2: continue

            overlap = len(k1 & k2) / min(len(k1), len(k2))
            if overlap >= 0.08:
                union(g1, g2)
                merged_pairs.append((g1, g2, round(overlap, 3)))
